// Gestures: the sound a soldier's gesture plays (TLK_NS_GESTURE, the gesture
// criterion nextsoldier.gestures' Name gives), TF2's, whoever the speaker:
// a body's sound, not a voice, played beside its lines (features/aiwar/
// sv_gestures.lua). A gesture with no rule here plays nothing.

criterion "IsDance" "gesture" "dance" required
criterion "IsMuscle" "gesture" "muscle" required
criterion "IsZombie" "gesture" "zombie" required
criterion "IsRobot" "gesture" "robot" required
criterion "IsLaugh" "gesture" "laugh" required

// The dance: the Halloween dance's howl, or Demo shaking it
response GestureDance
{
	speak "scout.boombox" nodelay
}
rule GestureDance
{
	criteria ConceptGesture IsDance
	response GestureDance
}

// Flexing: a chest thump, a strongman's rep, the yeti's chest hit
response GestureMuscle
{
	speak "scout.boombox"
}
rule GestureMuscle
{
	criteria ConceptGesture IsMuscle
	response GestureMuscle
}

// A taunt danced or flexed to the boombox (TLK_NS_TAUNT with the gesture,
// features/aiwar/sv_gestures.lua): the soldier cheers along in its own
// voice, in place of its long taunt. One rule a gesture, as a criterion
// matches one value; each outranks the voice's taunt by its gesture.
response CheerRebelMale
{
	speak "vo/npc/male01/yeah02.wav" defaultdelay
	speak "vo/npc/male01/nice.wav" defaultdelay
}
rule CheerRebelMale
{
	criteria ConceptTaunt IsRebel NotSaidTaunt IsMaleVoice IsDance
	ApplyContext "SaidTaunt:1:20"
	response CheerRebelMale
}
rule CheerRebelMaleFlexing
{
	criteria ConceptTaunt IsRebel NotSaidTaunt IsMaleVoice IsMuscle
	ApplyContext "SaidTaunt:1:20"
	response CheerRebelMale
}
